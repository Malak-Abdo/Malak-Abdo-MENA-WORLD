async function subscribe() { const name = document.getElementById('fullName').value.trim(); const email = document.getElementById('email').value.trim(); const position = getSelectValueWithOther('position', 'positionOther'); const country = getSelectValueWithOther('country', 'countryOther'); const city = getCityValue(); const organization = document.getElementById('organization').value.trim(); const linkedin = document.getElementById('linkedin').value.trim(); const consent = document.getElementById('consent').checked; const messageDiv = document.getElementById('registerMessage'); const subscribeBtn = document.getElementById('subscribeBtn'); // Validation if (!name) { showMessage(messageDiv, MSG.enterName, 'error-box'); return; } if (!email) { showMessage(messageDiv, MSG.enterEmail, 'error-box'); return; } if (!email.includes('@') || !email.includes('.')) { showMessage(messageDiv, MSG.validEmail, 'error-box'); return; } if (!position) { showMessage(messageDiv, MSG.enterPosition, 'error-box'); return; } if (!country) { showMessage(messageDiv, MSG.enterCountry, 'error-box'); return; } if (!city) { showMessage(messageDiv, MSG.enterCity, 'error-box'); return; } if (!organization) { showMessage(messageDiv, MSG.enterOrganization, 'error-box'); return; } if (!consent) { showMessage(messageDiv, MSG.noConsent, 'error-box'); return; } const userData = { name, email, position, country, city, organization, linkedin: linkedin || '' }; subscribeBtn.disabled = true; subscribeBtn.textContent = '⏳ Subscribing...'; showMessage(messageDiv, MSG.submitting, 'info-box'); // Save to Google Sheets ONLY — no email sent const saved = await saveToGoogleSheets(userData); if (saved) { showMessage(messageDiv, MSG.success, 'success-box'); localStorage.setItem('subscriberEmail', email); localStorage.setItem('subscriberName', name); resetForm(); } else { showMessage(messageDiv, MSG.saveError, 'error-box'); } subscribeBtn.disabled = false; subscribeBtn.textContent = '📨 Subscribe Now'; }
